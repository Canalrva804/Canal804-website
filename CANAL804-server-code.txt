const http=require('node:http'),fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');
const {DatabaseSync}=require('node:sqlite');
const PORT=Number(process.env.PORT||3000),BASE=__dirname,DB=process.env.DB_PATH||path.join(BASE,'store.sqlite');
const OWNER_EMAIL=(process.env.OWNER_LOGIN_EMAIL||'Canalrva804@gmail.com').trim().toLowerCase();const PASSWORD=process.env.OWNER_PASSWORD||'';if(PASSWORD.length<12) {console.error('Set OWNER_PASSWORD to a unique password at least 12 characters long.');process.exit(1)}
const db=new DatabaseSync(DB);db.exec(`PRAGMA journal_mode=WAL; CREATE TABLE IF NOT EXISTS threads(id TEXT PRIMARY KEY,token_hash TEXT UNIQUE NOT NULL,name TEXT,updated TEXT);CREATE TABLE IF NOT EXISTS messages(id INTEGER PRIMARY KEY AUTOINCREMENT,thread_id TEXT NOT NULL,sender TEXT NOT NULL,body TEXT NOT NULL,created TEXT NOT NULL);CREATE TABLE IF NOT EXISTS settings(key TEXT PRIMARY KEY,value TEXT);CREATE TABLE IF NOT EXISTS products(id TEXT PRIMARY KEY,name TEXT NOT NULL,description TEXT,price_cents INTEGER NOT NULL,image TEXT, status TEXT NOT NULL DEFAULT 'available',reserved_until INTEGER DEFAULT 0);`);
const q=(s,...a)=>db.prepare(s).get(...a), all=(s,...a)=>db.prepare(s).all(...a),run=(s,...a)=>db.prepare(s).run(...a);let sessions=new Map(),badLogins=new Map();const random=()=>crypto.randomBytes(32).toString('hex'),hash=v=>crypto.createHash('sha256').update(v).digest('hex'),now=()=>new Date().toISOString();
function headers(res,status,type='application/json',extras={}){res.writeHead(status,{'Content-Type':type,'Cache-Control':'no-store','X-Content-Type-Options':'nosniff','Content-Security-Policy':"default-src 'self' 'unsafe-inline'; img-src 'self' data: https:; connect-src 'self'; frame-src 'none'",...extras})}
function send(res,code,data){headers(res,code);res.end(JSON.stringify(data))}function cookie(req){let x=(req.headers.cookie||'').match(/(?:^|; )canal_admin=([^;]+)/);return x?decodeURIComponent(x[1]):''}function isOwner(req){let s=sessions.get(cookie(req));return !!s&&s>Date.now()}function requireOwner(req,res){if(!isOwner(req)){send(res,401,{error:'Owner sign-in required'});return false}return true}
async function body(req){let chunks=[],size=0;for await(const c of req){size+=c.length;if(size>250000)throw Error('Request too large');chunks.push(c)}return JSON.parse(Buffer.concat(chunks).toString()||'{}')}
const clean=(x,n=2000)=>typeof x==='string'?x.trim().slice(0,n):'';
async function handle(req,res){const u=new URL(req.url,'http://localhost');const m=req.method;try{
if(m==='GET'&&u.pathname==='/'){headers(res,200,'text/html; charset=utf-8',{'Cache-Control':'public,max-age=60'});return res.end(fs.readFileSync(path.join(BASE,'index.html')))}
if(m==='GET'&&u.pathname==='/api/status')return send(res,200,{ready:true,paymentMethod:'cash-in-person'});
if(m==='POST'&&u.pathname==='/api/messages'){let b=await body(req),message=clean(b.text),name=clean(b.name,80);if(!message)return send(res,400,{error:'Message cannot be empty'});let t=b.token&&q('SELECT * FROM threads WHERE token_hash=?',hash(String(b.token)));let raw=b.token;if(!t){raw=random();t={id:crypto.randomUUID()};run('INSERT INTO threads(id,token_hash,name,updated) VALUES(?,?,?,?)',t.id,hash(raw),name,now())}run('INSERT INTO messages(thread_id,sender,body,created) VALUES(?,?,?,?)',t.id,'visitor',message,now());run('UPDATE threads SET updated=? WHERE id=?',now(),t.id);return send(res,200,{token:raw})}
if(m==='GET'&&u.pathname==='/api/messages/thread'){let t=q('SELECT id FROM threads WHERE token_hash=?',hash(u.searchParams.get('token')||''));if(!t)return send(res,404,{error:'Conversation not found'});return send(res,200,{messages:all('SELECT sender,body AS text,created FROM messages WHERE thread_id=? ORDER BY id',t.id)})}
if(m==='POST'&&u.pathname==='/api/owner/login'){let ip=req.socket.remoteAddress||'unknown',st=badLogins.get(ip)||{n:0,until:0};if(st.until>Date.now())return send(res,429,{error:'Too many attempts; please try later'});let b=await body(req);let supplied=Buffer.from(String(b.password||'')),correct=Buffer.from(PASSWORD);let emailOK=typeof b.email==='string'&&b.email.trim().toLowerCase()===OWNER_EMAIL;if(!emailOK||supplied.length!==correct.length||!crypto.timingSafeEqual(supplied,correct)){st.n++;st.until=st.n>=5?Date.now()+15*60*1000:0;badLogins.set(ip,st);return send(res,401,{error:'Incorrect owner email or password'})}badLogins.delete(ip);let sid=random();sessions.set(sid,Date.now()+12*3600000);headers(res,200,'application/json',{'Set-Cookie':`canal_admin=${sid}; HttpOnly; SameSite=Strict; Path=/; Max-Age=43200${process.env.NODE_ENV==='production'?'; Secure':''}`});return res.end(JSON.stringify({ok:true}))}
if(m==='POST'&&u.pathname==='/api/owner/logout'){sessions.delete(cookie(req));headers(res,200,'application/json',{'Set-Cookie':'canal_admin=; HttpOnly; SameSite=Strict; Path=/; Max-Age=0'});return res.end('{"ok":true}')}
if(u.pathname.startsWith('/api/owner/')){if(!requireOwner(req,res))return;
if(m==='GET'&&u.pathname==='/api/owner/threads'){let ts=all('SELECT id,name,updated FROM threads ORDER BY updated DESC LIMIT 200');for(const t of ts)t.messages=all('SELECT sender,body AS text,created FROM messages WHERE thread_id=? ORDER BY id',t.id);return send(res,200,{threads:ts})}
if(m==='POST'&&u.pathname==='/api/owner/reply'){let b=await body(req),s=clean(b.text);if(!s||!q('SELECT id FROM threads WHERE id=?',b.id))return send(res,400,{error:'Unknown conversation or empty message'});run('INSERT INTO messages(thread_id,sender,body,created) VALUES(?,?,?,?)',b.id,'owner',s,now());run('UPDATE threads SET updated=? WHERE id=?',now(),b.id);return send(res,200,{ok:true})}
if(m==='POST'&&u.pathname==='/api/owner/products'){let b=await body(req),name=clean(b.name,120),desc=clean(b.description,500),cents=Math.round(Number(b.price)*100),image=clean(b.image,500);if(!name||!Number.isSafeInteger(cents)||cents<100||cents>100000000)return send(res,400,{error:'Enter a name and valid price of at least $1'});if(image&&!/^https:\/\//.test(image))return send(res,400,{error:'Image must use HTTPS'});let id=crypto.randomUUID();run('INSERT INTO products(id,name,description,price_cents,image,status) VALUES(?,?,?,?,?,?)',id,name,desc,cents,image,'available');return send(res,200,{id})}
if(m==='POST'&&u.pathname==='/api/owner/products/status'){let b=await body(req);if(!['available','pending','sold'].includes(b.status))return send(res,400,{error:'Invalid status'});let row=q('SELECT status FROM products WHERE id=?',b.id);if(!row)return send(res,409,{error:'Unavailable or reserved item'});run('UPDATE products SET status=?,reserved_until=0 WHERE id=?',b.status,b.id);return send(res,200,{ok:true})}
}
if(m==='GET'&&u.pathname==='/api/products'){return send(res,200,{products:all('SELECT id,name,description,price_cents AS priceCents,image,status FROM products ORDER BY rowid DESC')})}
return send(res,404,{error:'Not found'});
}catch(e){console.error('Request error:',e.message);return send(res,e.message==='Request too large'?413:500,{error:'Unable to complete request. Please try again.'})}}
http.createServer(handle).listen(PORT,()=>console.log('CANAL 804 server listening on '+PORT));
